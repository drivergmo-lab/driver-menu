// ========================================================== // ★ スマホ用：プルダウンの表示と内部状態を同期 // ========================================================== function syncDriverFromSelect() { const driverKey = els.driverSelect.value; if (!driverKey) { return false; } const driver = findDriverByKey( driverKey ); if (!driver) { return false; } /* * ここが重要。 * * スマホではプルダウンに * 「市川悟」と表示されていても * changeイベントが発生しないことがある。 * * その場合でもここで * state.driver を設定する。 */ state.driver = driver; /* * 自動判別中でも、 * ユーザーが選んだ人物として扱う。 */ state.userSelectedManually = true; enableAttendance( driver ); setStatus( 'done' ); return true; }
