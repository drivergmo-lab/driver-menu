// ============================================================ // 次回出勤日の表現を自動判定 // // 1日後 → 明日 // 2日後 → 明後日 // 次の週 → 来週 // 次の月 → 来月 // その他 → 次回 // ============================================================ function getNextAttendanceLabel( today, nextDate ) { const todayDate = new Date( today.getFullYear(), today.getMonth(), today.getDate() ); const targetDate = new Date( nextDate.getFullYear(), nextDate.getMonth(), nextDate.getDate() ); // 日数差 const diffMs = targetDate.getTime() - todayDate.getTime(); const diffDays = Math.round( diffMs / (1000 * 60 * 60 * 24) ); // ---------------------------------------------------------- // 明日 // ---------------------------------------------------------- if ( diffDays === 1 ) { return '明日'; } // ---------------------------------------------------------- // 明後日 // ---------------------------------------------------------- if ( diffDays === 2 ) { return '明後日'; } // ---------------------------------------------------------- // 来週 // // 月〜日を1週間として判定 // ---------------------------------------------------------- const todayDay = todayDate.getDay(); const targetDay = targetDate.getDay(); // 日曜=0 → 月曜始まりに変換 const todayWeekday = todayDay === 0 ? 7 : todayDay; const targetWeekday = targetDay === 0 ? 7 : targetDay; // 今週の月曜日 const thisWeekMonday = new Date( todayDate ); thisWeekMonday.setDate( todayDate.getDate() - ( todayWeekday - 1 ) ); // 次週の月曜日 const nextWeekMonday = new Date( thisWeekMonday ); nextWeekMonday.setDate( thisWeekMonday.getDate() + 7 ); // 対象日が次週 const targetWeekMonday = new Date( targetDate ); targetWeekMonday.setDate( targetDate.getDate() - ( targetWeekday - 1 ) ); if ( targetWeekMonday.getTime() === nextWeekMonday.getTime() ) { return '来週'; } // ---------------------------------------------------------- // 来月 // ---------------------------------------------------------- const nextMonth = new Date( todayDate.getFullYear(), todayDate.getMonth() + 1, 1 ); const targetMonth = new Date( targetDate.getFullYear(), targetDate.getMonth(), 1 ); if ( targetMonth.getTime() === nextMonth.getTime() ) { return '来月'; } // ---------------------------------------------------------- // その他 // ---------------------------------------------------------- return '次回'; } // ============================================================ // 退勤メッセージ // ============================================================ function buildOutMessage() { const today = new Date(); let next = tomorrowDate(); // ---------------------------------------------------------- // 次回出勤日 // ---------------------------------------------------------- if ( els.nextDate.value ) { next = new Date( els.nextDate.value + 'T00:00:00' ); } // ---------------------------------------------------------- // 次回出勤時間 // ---------------------------------------------------------- const nextTime = els.nextTime.value || '08:00'; // ---------------------------------------------------------- // 「明日」「明後日」「来週」「来月」「次回」 // ---------------------------------------------------------- const nextLabel = getNextAttendanceLabel( today, next ); // ---------------------------------------------------------- // 退勤メッセージ // ---------------------------------------------------------- return ( 'お疲れ様でございます。\n\n' + '業務完了いたしました。\n\n' + dateSlash( today ) + '\n' + '稼働)' + ( els.editClockInTime.value || '00:00' ) + '-' + ( els.clockOutTime.value || '00:00' ) + '\n\n' + '本日は失礼させていただきます。\n\n' + '出勤) ' + dateLabel( next ) + ' ' + nextTime + '\n\n' + nextLabel + 'もよろしくお願いいたします。' ); }
